
CREATE TABLE PROJECTCAMPAIGN
( 
campaign_id NUMBER,
title VARCHAR2(255), 
start_date DATE NOT NULL, 
end_date DATE NOT NULL, 
funding_goal DECIMAL(19,4) NOT NULL,
description VARCHAR2(255) NOT NULL, 
CONSTRAINT campaign_pk PRIMARY KEY (campaign_id)
);


CREATE TABLE PROJECTUSER
( 
user_id NUMBER,
first_name VARCHAR2(255) NOT NULL, 
last_name VARCHAR2(255) NOT NULL, 
password_hash VARCHAR2(255) NOT NULL,  
role VARCAHR2(20) CHECK(role IN('user','admin'))
);

CREATE TABLE PROJECTTRANSACTION
( 
transaction_id NUMBER,
campaign_id NUMBER, 
user_id NUMBER, 
authorization_num VARCHAR2(255) NOT NULL,
payment_type VARCHAR2(255) NOT NULL,
payment_date DATE NOT NULL, 
amount DECIMAL(19,4) NOT NULL, CHECK(amount>0),
CONSTRAINT fk_campaign FOREIGN KEY (campaign_id) REFERENCES ProjectCampaign(campaign_id),
CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES ProjectUser(user_id),
CONSTRAINT pk_transaction PRIMARY KEY (transaction_id)
);

CREATE TABLE PROJECTEMAIL
( 
user_id NUMBER, 
email VARCHAR2(255) NOT NULL,
CONSTRAINT fk_useremail FOREIGN KEY (user_id) REFERENCES ProjectUser(user_id),
CONSTRAINT pk_email PRIMARY KEY (user_id, email)
);

CREATE TABLE PROJECTSOCIALMEDIA
( 
user_id NUMBER, 
socialmedia VARCHAR2(255) NOT NULL,
CONSTRAINT fk_usersocialmedia FOREIGN KEY (user_id) REFERENCES ProjectUser(user_id),
CONSTRAINT pk_socialmedia PRIMARY KEY (user_id, socialmedia)
);

CREATE TABLE PROJECTUSERCAMPAIGN
( 
campaign_id NUMBER,
user_id NUMBER,
CONSTRAINT fk_campaignuser FOREIGN KEY (campaign_id) REFERENCES ProjectCampaign(campaign_id),
CONSTRAINT fk_usercompaign FOREIGN KEY (user_id) REFERENCES ProjectUser(user_id),
CONSTRAINT pk_ PRIMARY KEY (campaign_id, user_id)
);
